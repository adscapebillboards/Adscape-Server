const authenticateToken = (req, res, next) => {
    const authHeader = req.headers['authorization'];
    const token = authHeader?.split(' ')[1];
    if (!token) return res.status(401).json({ error: 'No token provided' });
  
    jwt.verify(token, JWT_SECRET, (err, user) => {
      if (err) return res.status(403).json({ error: 'Invalid token' });
      req.user = user;
      next();
    });
  };

app.post('/api/create-campaign', upload.array('files'), async (req, res) => {
  try {
    const { userName, billboards } = JSON.parse(req.body.data);
    const uploadedFiles = req.files;
    console.log('Received files:', userName,billboards); // 👀
    

    const campaignId = uuidv4(); // ✅ Generate unique ID

    const enrichedBillboards = await Promise.all(billboards.map(async (billboard) => {
      const { id, location, city, pricePerDay, bookingDetails,owner,screen_id } = billboard;
      const { startDate, endDate } = bookingDetails;

      const fileUrls = [];

      const matchingFiles = uploadedFiles.filter(file =>
        file.originalname.startsWith(`${id}_`)
      );

      const streamUpload = (fileBuffer) => {
        return new Promise((resolve, reject) => {
          const stream = cloudinary.uploader.upload_stream(
            { resource_type: 'image' },
            (error, result) => {
              if (error) return reject(error);
              resolve(result.secure_url);
            }
          );
          stream.end(fileBuffer);
        });
      };

      for (const file of matchingFiles) {
        try {
          const url = await streamUpload(file.buffer);
          fileUrls.push(url);
        } catch (error) {
          console.error(`❌ Failed to upload file ${file.originalname}:`, error.message);
        }
      }

      return {
        id,
        location,
        city,
        pricePerDay,
        bookingDetails: {
          startDate,
          endDate
        },
        files: fileUrls,
        owner,
        screen_id,
        // Campaign-related information for each billboard
        userName: userName,
        status: "PENDING",
        createDate: new Date().toISOString(),
        endDate: endDate,
        billboardCampaignId: `${campaignId}_${id}`, // Generate unique billboard campaign ID
        // Keep all existing billboard details
        ...billboard
      };
    }));

    const startDate = enrichedBillboards[0]?.bookingDetails.startDate;
    const endDate = enrichedBillboards[0]?.bookingDetails.endDate;

    // Calculate totalAmount
    const totalAmount = enrichedBillboards.reduce((sum, b) => {
      const days = (new Date(b.bookingDetails.endDate) - new Date(b.bookingDetails.startDate)) / (1000 * 60 * 60 * 24) + 1;
      return sum + (days * b.pricePerDay);
    }, 0);

    await pool.query(
      `INSERT INTO campaigns (id, user_name, campaign_name, status, total_amount, start_date, end_date, billboards)
       VALUES ($1, $2, $3, $4, $5, $6, $7, $8)`,
      [
        campaignId, // 👈 save the generated ID
        userName,
        "Auto Campaign",
        "pending",
        totalAmount,
        startDate,
        endDate,
        JSON.stringify(enrichedBillboards)
      ]
    );

    res.status(201).json({ 
      message: 'Campaign created successfully.',
      id: campaignId // 👈 send ID back in response
    });

  } catch (err) {
    console.error('❌ Error creating campaign:', err);
    res.status(500).json({ error: 'Internal server error.' });
  }
});

async function generateSlots(campaign) {
  const isTest= true
  const billboards = campaign.billboards;

  if (!Array.isArray(billboards)) {
    console.warn("❗ Billboards data missing or not an array");
    return;
  }

  for (const billboard of billboards) {
    const billboardId = billboard.id;
    const assetUrl = billboard.files?.[0];
    const screen_id = billboard.screen_id;
    const { startDate, endDate } = billboard.bookingDetails;

    if (!startDate || !endDate || !assetUrl) {
      console.warn(`⚠️ Missing data for billboard ${billboardId}`);
      continue;
    }

    const start = new Date(startDate);
    const end = new Date(endDate);

    // ✅ TEST SLOT: Generate slot on current date regardless of booking period
    if (isTest) {
      const today = new Date();
      const dateStr = today.toISOString().slice(0, 10);

      const countResult = await pool.query(
        `SELECT COUNT(*) FROM generated_slots
         WHERE billboard_id = $1 AND DATE(start_date) = $2`,
        [billboardId, dateStr]
      );

      const slotCount = parseInt(countResult.rows[0].count);

      if (slotCount < 8) {
        const slotId = uuidv4();
        const slotNumber = slotCount + 1;
        const slotStart = new Date(`${dateStr}T00:00:00Z`);
        const slotEnd = new Date(`${dateStr}T23:59:59Z`);

        await pool.query(
          `INSERT INTO generated_slots 
           (id, campaign_id, billboard_id, asset_url, start_date, end_date, duration, slot_number, screen_id)
           VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)`,
          [
            slotId,
            campaign.id,
            billboardId,
            assetUrl,
            slotStart,
            slotEnd,
            1,
            slotNumber,
            screen_id
          ]
        );

        console.log(`✅ TEST slot #${slotNumber} created for ${billboardId} on ${dateStr}`);
      } else {
        console.log(`⛔ TEST skipped: ${billboardId} already has 8 slots today`);
      }
    }

    // 🔁 Real campaign slot generation logic
    for (
      let current = new Date(start);
      current <= end;
      current.setDate(current.getDate() + 1)
    ) {
      const dateStr = current.toISOString().slice(0, 10);

      const countResult = await pool.query(
        `SELECT COUNT(*) FROM generated_slots
         WHERE billboard_id = $1 AND DATE(start_date) = $2`,
        [billboardId, dateStr]
      );

      const slotCount = parseInt(countResult.rows[0].count);

      if (slotCount >= 8) {
        console.log(`⛔ Skipped: ${billboardId} already has 8 slots on ${dateStr}`);
        continue;
      }

      const slotId = uuidv4();
      const slotNumber = slotCount + 1;

      const slotStart = new Date(`${dateStr}T00:00:00Z`);
      const slotEnd = new Date(`${dateStr}T23:59:59Z`);

      await pool.query(
        `INSERT INTO generated_slots 
         (id, campaign_id, billboard_id, asset_url, start_date, end_date, duration, slot_number, screen_id)
         VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)`,
        [
          slotId,
          campaign.id,
          billboardId,
          assetUrl,
          slotStart,
          slotEnd,
          1,
          slotNumber,
          screen_id
        ]
      );

      console.log(`🆕 Slot #${slotNumber} for ${billboardId} on ${dateStr}`);
    }
  }
}

app.post("/api/track-play", async (req, res) => {
  const { screen_id, asset_url, played_at } = req.body;

  try {
    const timestamp = played_at ? new Date(played_at) : new Date();
    const playDate = timestamp.toISOString().split("T")[0]; // YYYY-MM-DD

    // 🔍 Fetch campaign_id from generated_slots
    const campaignResult = await pool.query(
      `SELECT campaign_id FROM generated_slots
       WHERE screen_id = $1 AND asset_url = $2
       AND DATE(start_date) <= $3 AND DATE(end_date) >= $3
       LIMIT 1`,
      [screen_id, asset_url, playDate]
    );

    const campaign_id = campaignResult.rows[0]?.campaign_id || null;

    // 📝 Save detailed play log
    await pool.query(
      `INSERT INTO asset_play_logs (screen_id, asset_url, campaign_id, played_at)
       VALUES ($1, $2, $3, $4)`,
      [screen_id, asset_url, campaign_id, timestamp]
    );

    // 🔄 Update daily count
    await pool.query(
      `INSERT INTO asset_plays (screen_id, asset_url, campaign_id, play_date, play_count)
       VALUES ($1, $2, $3, $4, 1)
       ON CONFLICT (screen_id, asset_url, campaign_id, play_date)
       DO UPDATE SET play_count = asset_plays.play_count + 1`,
      [screen_id, asset_url, campaign_id, playDate]
    );

    res.sendStatus(200);
  } catch (err) {
    console.error("❌ DB error:", err);
    res.status(500).send("Internal Server Error");
  }
});

app.get("/api/campaign-metrics/:campaignId", async (req, res) => {
  const { campaignId } = req.params;

  try {
    // Fetch total displays per day for this campaign
    const [daily, weekly, monthly, total, perHour, peakHours] = await Promise.all([
      // 📅 Daily displays (Today only)
      pool.query(`
        SELECT SUM(play_count)::int AS count
        FROM asset_plays
        WHERE campaign_id = $1 AND play_date = CURRENT_DATE
      `, [campaignId]),

      // 📆 Weekly displays (Last 7 days)
      pool.query(`
        SELECT SUM(play_count)::int AS count
        FROM asset_plays
        WHERE campaign_id = $1 AND play_date >= CURRENT_DATE - INTERVAL '7 days'
      `, [campaignId]),

      // 🗓️ Monthly displays (Last 30 days)
      pool.query(`
        SELECT SUM(play_count)::int AS count
        FROM asset_plays
        WHERE campaign_id = $1 AND play_date >= CURRENT_DATE - INTERVAL '30 days'
      `, [campaignId]),

      // 📊 Total displays for campaign
      pool.query(`
        SELECT SUM(play_count)::int AS count
        FROM asset_plays
        WHERE campaign_id = $1
      `, [campaignId]),

      // 🕐 Average per hour (last 24h)
      pool.query(`
        SELECT COUNT(*) / 24.0 AS avg_per_hour
        FROM asset_play_logs
        WHERE campaign_id = $1 AND played_at >= NOW() - INTERVAL '1 day'
      `, [campaignId]),

      // ⏱️ Peak hours
      pool.query(`
        SELECT EXTRACT(HOUR FROM played_at) AS hour, COUNT(*)::int AS displays
        FROM asset_play_logs
        WHERE campaign_id = $1 AND played_at >= NOW() - INTERVAL '7 days'
        GROUP BY hour ORDER BY hour
      `, [campaignId]),
    ]);

    res.json({
      lastUpdated: new Date().toISOString(),
      dailyDisplays: daily.rows[0].count || 0,
      weeklyDisplays: weekly.rows[0].count || 0,
      monthlyDisplays: monthly.rows[0].count || 0,
      totalDisplays: total.rows[0].count || 0,
      averageDisplaysPerHour: Math.round(perHour.rows[0].avg_per_hour || 0),
      peakHours: peakHours.rows.map(row => ({
        hour: parseInt(row.hour),
        displays: row.displays,
      })),
    });
  } catch (err) {
    console.error("❌ Error in campaign metrics:", err);
    res.status(500).send("Server Error");
  }
});

app.get('/availability', async (req, res) => {
  const { state, city } = req.query;
  try {
    let result;
    if (city) {
      result = await pool.query('SELECT 1 FROM billboards WHERE city = $1 LIMIT 1', [city]);
    } else if (state) {
      result = await pool.query('SELECT 1 FROM billboards WHERE state = $1 LIMIT 1', [state]);
    }
    res.json({ available: result.rows.length > 0 });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Server error' });
  }
});
